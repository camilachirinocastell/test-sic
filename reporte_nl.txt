          56  def _parse_discount_pct(value):
          57      """Turn a string like '10% OFF' into the float 10.0. None if missing/unparseable."""
          58      if pd.isna(value):
          59          return None
          60      match = re.search(r"(\d+(?:\.\d+)?)", str(value))
          61      return float(match.group(1)) if match else None
          64  def _to_nullable_bool(value):
          65      """Turn pandas True/False/NaN into 1/0/None for SQLite storage."""
          66      if pd.isna(value):
          67          return None
          68      return 1 if bool(value) else 0
          71  def load_csv_to_db(csv_path: str = CSV_PATH, db_path: str = DATABASE_PATH) -> int:
          72      """
          73      Read the source CSV, map/clean its columns according to SCHEMA, and
          74      load it into a fresh SQLite table. Meant to run once at startup, not
          75      on every query.
          77      Returns the number of rows loaded.
          78      Raises CSVLoadError if the file is missing, empty, malformed, or
          79      missing a column that SCHEMA expects.
          80      """
          81      path = Path(csv_path)
          82      if not path.exists():
          83          raise CSVLoadError(f"CSV file not found at '{csv_path}'")
          85      try:
          86          df = pd.read_csv(path)
          87      except pd.errors.EmptyDataError:
          88          raise CSVLoadError(f"CSV file at '{csv_path}' is empty")
          89      except pd.errors.ParserError as exc:
          90          raise CSVLoadError(f"CSV file at '{csv_path}' is malformed: {exc}")
          92      if df.empty:
          93          raise CSVLoadError(f"CSV file at '{csv_path}' has no rows")
          95      missing = [info["source"] for info in SCHEMA.values() if info["source"] not in df.columns]
          96      if missing:
          97          raise CSVLoadError(f"CSV is missing expected columns: {missing}")
          99      clean = pd.DataFrame()
         100      for final_name, info in SCHEMA.items():
         101          source_col = info["source"]
         102          if final_name == "discount_pct":
         103              clean[final_name] = df[source_col].apply(_parse_discount_pct)
         104          elif final_name in ("free_shipping", "official_store", "in_stock"):
         105              clean[final_name] = df[source_col].apply(_to_nullable_bool)
         106          else:
         107              clean[final_name] = df[source_col]
         109      if clean["item_id"].duplicated().any():
         111          clean = clean.drop_duplicates(subset="item_id", keep="first")
         113      Path(db_path).parent.mkdir(parents=True, exist_ok=True)
         114      conn = sqlite3.connect(db_path)
         115      try:
         116          columns_sql = ", ".join(f"{name} {info['sql_type']}" for name, info in SCHEMA.items())
         117          conn.execute(f"DROP TABLE IF EXISTS {TABLE_NAME}")
         118          conn.execute(f"CREATE TABLE {TABLE_NAME} ({columns_sql})")
         119          clean.to_sql(TABLE_NAME, conn, if_exists="append", index=False)
         120          conn.commit()
         121      finally:
         122          conn.close()
         125      return len(clean)

         139  def execute_query(sql: str, db_path: str = DATABASE_PATH) -> dict:
         140      """
         141      Execute a raw SQL string and return {"columns": [...], "rows": [...]}.
         143      NOTE: this does NOT validate or sanitize the SQL yet — that's
         144      explicitly deferred to Phase 2 (see section 5.1: SQL injection
